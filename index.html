window.generateAdvice = async () => { try { const prompt = document.getElementById("prompt").value; if (!prompt.trim()) { alert("내용을 입력하세요."); return; } const GEMINI_API_KEY = "여기에_실제_API키"; document.getElementById("result").innerText = "AI가 분석 중입니다..."; const response = await fetch( `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${GEMINI_API_KEY}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [ { parts: [ { text: ` 당신은 경영 컨설턴트입니다. 다음 상황을 분석하고 1. 문제 원인 2. 해결 방안 3. 마케팅 전략 4. 예상 효과 를 한국어로 작성하세요. ${prompt} ` } ] } ] }) } ); const data = await response.json(); console.log("Gemini 응답:", data); if (!response.ok) { alert( "Gemini 오류:\n" + JSON.stringify(data, null, 2) ); return; } const text = data.candidates?.[0]?.content?.parts?.[0]?.text; if (!text) { alert( "AI 응답이 비어있습니다.\n콘솔(F12)을 확인하세요." ); return; } document.getElementById("result").innerText = text; } catch (error) { console.error(error); alert( "오류 발생:\n" + error.message ); } };
